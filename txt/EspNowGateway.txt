/**
 * @file gateway_main.cpp
 * @brief ESP32 Gateway — ESP-NOW <-> MQTT (Native Pure Version)
 *
 * External Dependencies: PubSubClient (knolleary)
 * No custom libraries.
 *
 * Fixes applied:
 *   1. WiFi.setSleep(WIFI_PS_NONE) -> Radio always active, no ACK loss
 *   2. Native learnPeer()          -> Peer registered with correct channel
 *   3. ANNOUNCE (type 0x01)        -> Only learning, not forwarded to MQTT
 *   4. All other types             -> Published raw to MQTT
 *   5. ISR-safe FIFO               -> ESP-NOW callback doesn't block the loop
 */

#include "esp_bt.h"
#include "esp_bt_main.h"
#include "esp_wifi.h"
#include <Arduino.h>
#include <PubSubClient.h>
#include <WiFi.h>
#include <esp_now.h>

#include <password.h>
#include <shared_config.h>

/* ========================================================================= *
 *                              CONFIGURATION                                *
 * ========================================================================= */

// PacketProtocol Types
#define PKT_ANNOUNCE 0x01  // Learning only — DO NOT publish to MQTT
#define CMD_GET_NODES 0x90 // Command to request node list

// MAC Addresses
uint8_t broadcastAddress[] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

/* ========================================================================= *
 *                            DATA STRUCTURES                                *
 * ========================================================================= */

// Response packet structure
struct struct_risposta {
  char messaggio[32];
  int status;
};

// Global response instance
struct_risposta risposta;

// FIFO settings
#define FIFO_DEPTH 8
#define MAX_PKT_SIZE 250

struct RxSlot {
  uint8_t mac[6];
  uint8_t data[MAX_PKT_SIZE];
  uint8_t len;
  bool used;
};

/* ========================================================================= *
 *                            GLOBAL VARIABLES                               *
 * ========================================================================= */

static RxSlot s_fifo[FIFO_DEPTH] = {};
static volatile uint8_t s_head = 0;
static volatile uint8_t s_tail = 0;

// WiFiClient wifiClient;
// PubSubClient mqtt(wifiClient);
const char *PROGETTO_TOPICS[] = {
    systemTopic,     updateTopic, espNowBridgeCmd, espNowBridgeBuffer,
    espNowBridgeAck, // ← Forward ACK da mariner.js alla sonda via ESP-NOW
    nullptr          // OBBLIGATORIO: sentinella di fine array
};
MotivoSpegnimento m_wifi_status;

// MAC dell'ultimo mittente ESP-NOW che ha inviato dati (non announce)
// Usato per girare l'ACK di mariner.js alla sonda corretta.
static uint8_t s_last_data_sender_mac[6] = {0};
static bool s_last_data_sender_valid = false;

/* ========================================================================= *
 *                            NODE TRACKING                                  *
 * ========================================================================= */

struct NodeRecord {
  uint8_t deviceID;
  uint8_t mac[6];
  uint32_t lastSeen;
  uint16_t fwVersion;
  bool isEspNow;
};

static NodeRecord s_nodeTable[32];
static int s_nodeTableSize = 0;

/**
 * Periodically removes nodes not seen for more than 1 hour.
 */
static void cleanupNodeTable() {
  static uint32_t lastCleanup = 0;
  if (millis() - lastCleanup < 60000)
    return; // Run cleanup once a minute
  lastCleanup = millis();

  uint32_t now = millis();
  for (int i = 0; i < s_nodeTableSize; i++) {
    if (now - s_nodeTable[i].lastSeen > 3600000) { // 1 hour
      LOG_INFO("[SYSTEM] Rimozione nodo inattivo: 0x%02X",
               s_nodeTable[i].deviceID);
      for (int j = i; j < s_nodeTableSize - 1; j++) {
        s_nodeTable[j] = s_nodeTable[j + 1];
      }
      s_nodeTableSize--;
      i--;
    }
  }
}

/**
 * Updates the local table of known nodes.
 */
static void updateNodeTable(uint8_t devID, const uint8_t mac[6], bool isEspNow,
                            uint16_t fwVer = 0) {
  int idx = -1;
  for (int i = 0; i < s_nodeTableSize; i++) {
    if (s_nodeTable[i].deviceID == devID) {
      idx = i;
      break;
    }
  }

  if (idx == -1) {
    if (s_nodeTableSize < 32) {
      idx = s_nodeTableSize++;
    } else {
      idx = 0;
      for (int i = 1; i < 32; i++) {
        if (s_nodeTable[i].lastSeen < s_nodeTable[idx].lastSeen)
          idx = i;
      }
    }
    s_nodeTable[idx].deviceID = devID;
    s_nodeTable[idx].fwVersion = 0;
    s_nodeTable[idx].isEspNow = false; // Initial assumption
    memset(s_nodeTable[idx].mac, 0, 6);
    LOG_INFO("[SYSTEM] Nuovo nodo registrato: 0x%02X", devID);
  }

  uint32_t now = millis();
  if (fwVer != 0)
    s_nodeTable[idx].fwVersion = fwVer;

  if (isEspNow) {
    s_nodeTable[idx].isEspNow = true;
    s_nodeTable[idx].lastSeen = now;
    if (mac)
      memcpy(s_nodeTable[idx].mac, mac, 6);
  } else {
    // Via MQTT: skip update if we saw it via ESP-NOW in the last 2s (loopback)
    if (s_nodeTable[idx].isEspNow && (now - s_nodeTable[idx].lastSeen < 2000)) {
        return;
    }
    s_nodeTable[idx].isEspNow = false;
    s_nodeTable[idx].lastSeen = now;
  }
}

/**
 * Publishes a JSON report of all tracked nodes to MQTT.
 */
static void sendNodeReport() {
  String report = "{\"gateway\":\"" + WiFi.macAddress() + "\",\"nodes\":[";
  uint32_t now = millis();
  for (int i = 0; i < s_nodeTableSize; i++) {
    if (i > 0)
      report += ",";
    char buf[160];
    uint32_t ago = (now - s_nodeTable[i].lastSeen) / 1000;
    snprintf(
        buf, sizeof(buf),
        "{\"id\":\"0x%02X\",\"type\":\"%s\",\"fw_version\":%u,\"last_seen_s\":%"
        "u,\"mac\":\"%02X:%02X:%02X:%02X:%02X:%02X\"}",
        s_nodeTable[i].deviceID, s_nodeTable[i].isEspNow ? "ESP-NOW" : "MQTT",
        s_nodeTable[i].fwVersion, ago, s_nodeTable[i].mac[0],
        s_nodeTable[i].mac[1], s_nodeTable[i].mac[2], s_nodeTable[i].mac[3],
        s_nodeTable[i].mac[4], s_nodeTable[i].mac[5]);
    report += buf;
  }
  report += "]}";
  bool ok = mqttWifi::client.publish(logTopic, report.c_str());
  mqttWifi::loop(); // Ensure the report is sent immediately
  if (ok) {
    LOG_INFO("[SYSTEM] Node report published to %s (%d bytes)", logTopic, report.length());
  } else {
    LOG_ERROR("[SYSTEM] FAILED to publish node report (size too large or MQTT disconnected!)");
  }
}
/* ========================================================================= *
 *                            FIFO HELPERS                                   *
 * ========================================================================= */

static inline bool fifoFull() { return ((s_tail + 1) % FIFO_DEPTH) == s_head; }

static inline bool fifoEmpty() { return s_head == s_tail; }

/* ========================================================================= *
 *                            IMPLEMENTATION                                 *
 * ========================================================================= */

/**
 * Registers the source MAC as an ESP-NOW peer (idempotent).
 * Necessary because ESP32 in WIFI_STA accepts unicast only from registered
 * peers.
 */
static void learnPeer(const uint8_t *mac) {
  if (esp_now_is_peer_exist(mac)) {
    return;
  }

  esp_now_peer_info_t peer = {};
  memcpy(peer.peer_addr, mac, 6);
  peer.channel = (uint8_t)WiFi.channel(); // MUST match the AP channel
  peer.encrypt = false;
  // FONDAMENTALE: Usare l'interfaccia AP, altrimenti la modalità STA filtra e
  // sega gli Unicast in ingresso!
  peer.ifidx = WIFI_IF_AP;

  esp_err_t err = esp_now_add_peer(&peer);
  LOG_INFO("[LEARN] Peer %02X:%02X:%02X:%02X:%02X:%02X -> %s (ch %d)", mac[0],
           mac[1], mac[2], mac[3], mac[4], mac[5],
           err == ESP_OK ? "OK" : esp_err_to_name(err), WiFi.channel());
}

/**
 * ESP-NOW Receive Callback
 * ISR-like context: ONLY copies to FIFO, nothing else.
 */
static void onEspNowRecv(const uint8_t *mac, const uint8_t *data, int len) {
  // 1. Validazione rapida
  if (!mac || !data || len <= 0 || len > MAX_PKT_SIZE)
    return;

  // 2. Se la FIFO è piena, usciamo subito
  if (fifoFull())
    return;

  // 3. Copia i dati nella FIFO (Operazione velocissima)
  RxSlot &slot = s_fifo[s_tail];
  memcpy(slot.mac, mac, 6);
  memcpy(slot.data, data, len);
  slot.len = (uint8_t)len;
  slot.used = true;
  s_tail = (s_tail + 1) % FIFO_DEPTH;

  // NON INVIARE RISPOSTE QUI. Falle nel loop dopo aver letto dalla FIFO.
}

/**
 * ESP-NOW Send Callback
 * Hardware-level confirmation (ACK)
 */
static void onEspNowSend(const uint8_t *mac, esp_now_send_status_t status) {
  if (status == ESP_NOW_SEND_SUCCESS) {
    LOG_VERBOSE("[TX-ACK] Consegnato a %02X:%02X:%02X:%02X:%02X:%02X", mac[0],
                mac[1], mac[2], mac[3], mac[4], mac[5]);
  } else {
    LOG_WARN("[TX-ACK] FALLITO invio a %02X:%02X:%02X:%02X:%02X:%02X (Nodo "
             "sordo o canale errato)",
             mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  }
}

/**
 * MQTT Message Callback (Node-RED → nodes)
 * Receives an already formatted PacketProtocol packet from Node-RED
 * and forwards it to the correct node via ESP-NOW.
 */
static void onMqttMessage(char *topic, byte *payload, unsigned int length) {

  // --- CASO 1: ACK da mariner.js → girarlo alla sonda via ESP-NOW ---
  if (strcmp(topic, espNowBridgeAck) == 0) {
    if (!s_last_data_sender_valid) {
      LOG_WARN("[ACK-FWD] ACK ricevuto ma nessun mittente ESP-NOW registrato!");
      return;
    }
    // Copia il payload in un buffer null-terminated per il log
    char ackStr[16] = {0};
    size_t toCopy = (length < sizeof(ackStr) - 1) ? length : sizeof(ackStr) - 1;
    memcpy(ackStr, payload, toCopy);

    LOG_INFO("[ACK-FWD] Girato '%s' a %02X:%02X:%02X:%02X:%02X:%02X", ackStr,
             s_last_data_sender_mac[0], s_last_data_sender_mac[1],
             s_last_data_sender_mac[2], s_last_data_sender_mac[3],
             s_last_data_sender_mac[4], s_last_data_sender_mac[5]);

    // Invia la stringa grezza ("ack"/"end"/"failed"/"error") come si aspetta
    // waitForAck()
    esp_err_t err =
        esp_now_send(s_last_data_sender_mac, (const uint8_t *)ackStr, toCopy);
    if (err != ESP_OK) {
      LOG_ERROR("[ACK-FWD] esp_now_send fallito: %s", esp_err_to_name(err));
    }

    // Se l'ACK è finale ("end" o "error") resettiamo il mittente attivo
    if (strcmp(ackStr, "end") == 0 || strcmp(ackStr, "error") == 0) {
      s_last_data_sender_valid = false;
    }
    return;
  }

  // --- CASO 2: Comando o Dati in ingresso da MQTT → ESP-NOW (bridge inverso)
  // --- Inoltriamo alla radio tutto ciò che non è indirizzato specificamente al gateway
  if (strcmp(topic, espNowBridgeCmd) == 0 ||
      strcmp(topic, espNowBridgeBuffer) == 0) {
    if (length < 6 || payload[0] != 0xAA) {
      LOG_WARN("[MQTT] Pacchetto non valido (%d byte) su %s", length, topic);
      return;
    }

    uint8_t devId = payload[5];
    uint8_t type = payload[2];

    // Track node if it's on the buffer
    if (strcmp(topic, espNowBridgeBuffer) == 0) {
      if (devId != DEV_ESP_NOW_GATEWAY) {
        uint16_t fw = 0;
        if (type == PKT_ANNOUNCE && length >= 9) {
          fw = payload[7] | (payload[8] << 8);
        }
        updateNodeTable(devId, nullptr, false, fw);
      }
    }

    // BRIDGE INVERSO: Se il pacchetto è per altri nodi, rilanciamo ai nodi ESP-NOW conosciuti (Unicast Unroll)
    if (devId != DEV_ESP_NOW_GATEWAY) {
      LOG_VERBOSE("[MQTT->ESPNOW] Bridge FROM topic %s TO ESP-NOW (Unicast Unroll) | "
                  "Type: 0x%02X | DevID: 0x%02X",
                  topic, type, devId);

      bool sent_to_someone = false;
      for (int i = 0; i < s_nodeTableSize; i++) {
        if (s_nodeTable[i].isEspNow) {
          esp_err_t err = esp_now_send(s_nodeTable[i].mac, (const uint8_t *)payload, length);
          sent_to_someone = true;
          if (err != ESP_OK) {
             LOG_ERROR("[MQTT] esp_now_send fallito verso 0x%02X: %s", s_nodeTable[i].deviceID, esp_err_to_name(err));
          }
        }
      }

      if (!sent_to_someone) {
        LOG_WARN("[MQTT->ESPNOW] Nessun nodo ESP-NOW registrato. Uso Fallback Broadcast.");
        static const uint8_t broadcast[6] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};
        esp_now_send(broadcast, (const uint8_t *)payload, length);
      }
      return;
    }

    // Se arriviamo qui, il pacchetto è diretto al gateway (devId == DEV_ESP_NOW_GATEWAY)
    if (type == TYPE_COMMAND && length >= 7) {
      const cmdData *d = (const cmdData *)(payload + 5);
      if (d->command == CMD_GET_NODES) {
        LOG_VERBOSE("[MQTT] Comando GET_NODES ricevuto — invio report nodi");
        sendNodeReport();
      } else {
        LOG_WARN("[MQTT] Comando gateway sconosciuto 0x%02X", d->command);
      }
      return;
    }
  }
}


/* ========================================================================= *
 *                              MAIN ARDUINO                                 *
 * ========================================================================= */

void setup() {
  logSerialBegin(38400);
  delay(500);
  btStop();
  LOG_INFO("========================================");
  LOG_INFO("   ESP32 GATEWAY (UDP LOG ENABLED)      ");
  LOG_INFO("========================================");

  m_wifi_status =
      mqttWifi::setupCompleto(ipEspNowBridge, EspNowBridge, PROGETTO_TOPICS);
  if (m_wifi_status != CONN_OK) {
    LOG_ERROR("[WIFI] FALLITO — riavvio");
    delay(20000);
    ESP.restart();
  }

  // Abilita Log UDP dopo connessione WiFi
  udpLogBegin();
  logSetDeviceName("EspNowGateway");

  WiFi.mode(WIFI_AP_STA);
  // Avviamo il SoftAP sullo stesso canale della STA per stabilizzare la radio
  WiFi.softAP("ESP-NOW-GATEWAY", "12345678", WiFi.channel());
  LOG_INFO("[SYSTEM] Modalità WIFI_AP_STA con SoftAP attivo su ch %d",
           WiFi.channel());
  // IMPORTANT: Disable WiFi power save or ESP-NOW will lose packets
  esp_wifi_set_ps(WIFI_PS_NONE);

  mqttWifi::client.setCallback(onMqttMessage);
  mqttWifi::client.setBufferSize(2048); // Support reports for many nodes

  // 4. Initialize ESP-NOW (Must be done AFTER WiFi is connected/initialized)
  if (esp_now_init() != ESP_OK) {
    LOG_ERROR("[ESPNOW] Init FALLITO — riavvio");
    delay(2000);
    ESP.restart();
  }

  // Prepare the global response packet
  strcpy(risposta.messaggio, "ack");
  risposta.status = 200;

  // Register receiving callback
  esp_now_register_recv_cb(onEspNowRecv);
  esp_now_register_send_cb(onEspNowSend);

  // 5. REGISTER BROADCAST PEER
  esp_now_peer_info_t bcastPeer = {};
  memset(bcastPeer.peer_addr, 0xFF, 6); // Broadcast address

  // CRITICAL: The channel must be the one tuned by the WiFi
  // If the ESP-01 is on 12, your Router MUST be on 12,
  // otherwise the ESP32 won't hear it.
  bcastPeer.channel = WiFi.channel();
  bcastPeer.encrypt = false;
  bcastPeer.ifidx = WIFI_IF_AP;

  if (esp_now_add_peer(&bcastPeer) != ESP_OK) {
    LOG_ERROR("[ESPNOW] Errore aggiunta Peer Broadcast");
  }

  LOG_INFO("[SYSTEM] MAC Gateway: %s", WiFi.macAddress().c_str());
  LOG_INFO("[ESPNOW] In ascolto sul canale: %d", bcastPeer.channel);

  if (bcastPeer.channel != WIFI_CHANNEL_GATEWAY) {
    LOG_WARN("⚠️ ATTENZIONE: Il Gateway è su un canale diverso dal %d!",
             WIFI_CHANNEL_GATEWAY);
    LOG_WARN("Assicurati che il tuo Router Wi-Fi sia sul canale %d o "
             "l'ESP-01 non lo troverà.",
             WIFI_CHANNEL_GATEWAY);
  } else {
    LOG_INFO("[SYSTEM] Canale Wi-Fi corretto per ESP-NOW: %d",
             bcastPeer.channel);
  }
}

void loop() {
  // 3. WiFi Watchdog (every 30s)
  static uint32_t lastWifiCheck = 0;
  if (millis() - lastWifiCheck > 30000) {
    lastWifiCheck = millis();
    LOG_VERBOSE("[ESPNOW] In ascolto sul canale: %d", WiFi.channel());
    m_wifi_status = mqttWifi::gestisciConnessione();
    if (m_wifi_status != CONN_OK) {
      LOG_ERROR("[WIFI] Connessione persa — riconnessione");
      delay(20000);
      ESP.restart();
    }
  }
  mqttWifi::loop();

  // Point 4: Stale nodes cleanup
  cleanupNodeTable();

  // 2. Consume ESP-NOW FIFO
  while (!fifoEmpty()) {
    RxSlot &slot = s_fifo[s_head];
    s_head = (s_head + 1) % FIFO_DEPTH;

    const uint8_t *mac = slot.mac;
    const uint8_t *data = slot.data;
    uint8_t len = slot.len;

    // Log del pacchetto grezzo
    LOG_INFO("[RX] %02X:%02X:%02X:%02X:%02X:%02X | Len: %d | Type: 0x%02X",
             mac[0], mac[1], mac[2], mac[3], mac[4], mac[5], len,
             len >= 3 ? data[2] : 0xFF);

    // Validazione minima del frame
    if (len < 5 || data[0] != 0xAA) {
      LOG_WARN("[RX] Frame non valido — scartato");
      continue;
    }

    uint8_t type = data[2];
    uint8_t devID = data[5];

    // --- LOGICA DI APPRENDIMENTO PEER ---
    learnPeer(mac);
    if (type != TYPE_COMMAND && type != TYPE_ACK) {
      updateNodeTable(devID, mac, true);
    }

    // Se è un comando diretto al gateway via ESP-NOW
    if (type == TYPE_COMMAND && len >= 7) {
      const cmdData *d = (const cmdData *)(data + 5);
      if (d->command == CMD_GET_NODES) {
        sendNodeReport();
      }
    }

    // --- CASO 1: ANNOUNCE (0x01) ---
    if (type == PKT_ANNOUNCE) {
      const announceData *ann = (const announceData *)(data + 5);
      updateNodeTable(devID, mac, true, ann->fwVersion);
      LOG_INFO("[RX] ANNOUNCE ricevuto da 0x%02X (fw %u)", devID, ann->fwVersion);

      // Risposta binaria v3 (TYPE_ACK)
      uint8_t ackBuf[128];
      ackData resp;
      resp.deviceID = devID;
      resp.status = AC_SWITCH_TO_ESPNOW;
      resp.cmdEcho = PKT_ANNOUNCE;
      resp.valEcho = 0;

      size_t sz = pp_buildPacket(0x00, (uint8_t *)&resp, sizeof(resp), ackBuf);
      esp_now_send(mac, ackBuf, sz); // <-- Cambiato da broadcastAddress a mac (UNICAST)
      continue; // Non pubblichiamo l'Announce su MQTT
    }

    // --- CASO 2: ALTRI PACCHETTI (DATI/STATUS/COMANDI) → MQTT ---
    if (m_wifi_status == CONN_OK) {
      // Salva il MAC del mittente per girare l'ACK di mariner.js
      memcpy(s_last_data_sender_mac, mac, 6);
      s_last_data_sender_valid = true;

      // Pubblicazione dati su MQTT
      bool ok = mqttWifi::client.publish(espNowBridgeBuffer, data, len, false);
      LOG_INFO("[RX] MQTT Publish su %s: %s", espNowBridgeBuffer, ok ? "OK" : "FALLITO");

      if (!ok) {
        s_last_data_sender_valid = false;
        const char *failMsg = "failed";
        esp_now_send(mac, (const uint8_t *)failMsg, strlen(failMsg));
      }

      // --- CASO 3: REBROADCAST verso altri nodi ESP-NOW ---
      // Necessario per Chrono2 e altri nodi che ascoltano pure ESP-NOW
      // Lo facciamo iterando sui nodi validi per evitare le perdite dei Broadcast
      for (int i = 0; i < s_nodeTableSize; i++) {
        if (s_nodeTable[i].isEspNow && memcmp(s_nodeTable[i].mac, mac, 6) != 0) {
           esp_now_send(s_nodeTable[i].mac, data, len);
        }
      }

    } else {
      LOG_WARN("[RX] MQTT non connesso — pacchetto scartato");
      const char *errMsg = "failed";
      esp_now_send(mac, (const uint8_t *)errMsg, strlen(errMsg));
    }
  }
}
