void goToDeepSleep() {
  LOG_INFO("[SLEEP] Vado in deep sleep per %d ms...\n", WORK_INTERVAL_MS);
  esp_sleep_enable_timer_wakeup(WORK_INTERVAL_MS);
  esp_deep_sleep_start();
}

void setup() {
  esp_sleep_wakeup_cause_t wakeup_reason = esp_sleep_get_wakeup_cause();
  btStop(); // Disabilita Bluetooth per risparmiare energia

  sensors::setup();

#ifdef TEST_DEBUG 
 #warning "TEST MODE ENABLED!"
  Serial.begin(38400);
  delay(500);
  LOG_VERBOSE("\n[BOOT] SENSOR VIVAIO (C3) - BOOT NUMERO: %d\n", bootCounter);
  if (wakeup_reason == ESP_SLEEP_WAKEUP_TIMER) {
    LOG_VERBOSE("[MAIN] Risvegliato da timer");
  } else {
    LOG_VERBOSE("[MAIN] Risvegliato da causa diversa");
  }
#endif

  // Se veniamo dal deep sleep, aspetta di più per far stabilizzare l'hardware
  if (wakeup_reason == ESP_SLEEP_WAKEUP_TIMER) {
    delay(500); 
  } else {
    delay(100);
  }

  bootCounter++;

  // === 1. LEGGI SENSORI ===
  uint16_t soilRaw, soilPercent;
  uint16_t batteryMv;
  int16_t tempBmp;

  bool soilOk = sensors::readSoilMoisture(soilRaw, soilPercent);
  if (!soilOk){
    LOG_WARN("[MAIN] Lettura umidità su ADC fallita!");
    soilPercent = 0xFFFE; // Valore speciale per indicare errore (0xFFFF è 100%)
  }

  if (!sensors::readBatteryVoltage(batteryMv)) {
    LOG_WARN("[MAIN] Lettura batteria su ADC fallita!");
    batteryMv = 0;
  }
  
  bool tempOk = sensors::readTemperature(tempBmp);
  if (!tempOk) LOG_WARN("[MAIN] Lettura temperatura fallita!");

  // === 2. INIZIALIZZA ESP-NOW TRAMITE ASTRAZIONE ===
  mqttWifi::m_deviceID = DEVICE_ID; // Importante per il pacchetto di Announce
  if (!mqttWifiMessages::initTransport()) {
      LOG_ERROR("[MAIN] Impossibile inizializzare il trasporto ESP-NOW!\n");
      WiFi.mode(WIFI_OFF);
    goToDeepSleep();
  }

  // === 3. INVIA DATI ===
  bool success = mqttWifiMessages::sendData(DEVICE_ID, soilRaw, batteryMv, tempBmp, bootCounter);

  // === 4. INVIO SALUTE SENSORI (Tier 4 protocol ONE-SHOT) ===
  // Bitmask: bit0=soil OK, bit1=DS18B20 OK
  // Il gateway aggiorna la nodeTable senza che il nodo sia registrato.
  uint8_t healthMask = (soilOk ? 0x01 : 0x00) | (tempOk ? 0x02 : 0x00);
  mqttWifiMessages::sendHealthAck(DEVICE_ID, healthMask);

  // === 5. HEARTBEAT LED ===
  if (success) {
    LOG_INFO("[MAIN] Dati inviati con successo! Vado a dormire...\n");
    sensors::blinkLED(1, 100);
  } else {
    LOG_ERROR("[MAIN] Errore nell'invio dati!\n");
    sensors::blinkLED(3, 50);
  }

  // === 6. SLEEP ===
  if (mqttWifiMessages::transport) {
    mqttWifiMessages::transport->disconnect(); // Deinizializza ESP-NOW e libera peer
  }
  WiFi.mode(WIFI_OFF);
  goToDeepSleep();
}

void loop() {


  goToDeepSleep();
}

/**
 * @file        mqttWifiMessages.cpp
 * @project     ESP32C3 Piante - Soil Sensor
 * @brief       Livello messaggi per nodo ONE-SHOT via ESP-NOW.
 *
 *              Strategia: NON viene eseguita la registrazione del nodo
 *              (ANNOUNCE handshake). Il nodo si sveglia, inizializza
 *              ESP-NOW con il solo peer broadcast, invia i propri dati
 *              e l'ACK di salute (Tier 4 ONE-SHOT), poi va in sleep.
 *
 *              Modello: Fire-and-Forget (Telemetria Passiva, QoS 0).
 */
#include "mqttWifiMessages.h"
#include "impostazioni.h"

namespace mqttWifiMessages {

INetworkTransport* transport = nullptr;

/**
 * @brief Inizializza ESP-NOW senza ANNOUNCE handshake.
 *        Per i nodi one-shot non è necessaria la registrazione:
 *        il gateway li tratta come sorgenti passive.
 */
bool initTransport() {
    transport = createNetworkTransport(NetworkTransportType::ESPNOW);
    if (!transport) return false;
    // Solo init(): aggiunge il peer broadcast e predispone la radio.
    // NON chiamiamo connect() che farebbe l'ANNOUNCE handshake.
    return transport->init();
}

/**
 * @brief Invia i dati del sensore in broadcast (fire & forget).
 *        Nessuna attesa ACK — il gateway li riceve passivamente.
 */
bool sendData(uint8_t deviceId, uint16_t moisture, uint16_t battery,
              int16_t temperatureBmp, uint8_t counter) {
    if (!transport) return false;

    struct meteoData m;
    memset(&m, 0, sizeof(m));
    m.deviceID        = deviceId;
    m.moisture        = moisture;
    m.battery         = battery;
    m.counter         = counter;
    m.humidityBMP     = 0;
    m.temperatureBMP  = temperatureBmp;
    m.externalPressure = 0;

    m.checksum = pp_calculateXOR((uint8_t *)&m, sizeof(m) - 1);

    uint8_t outPacket[64];
    outPacket[0] = PACKET_MAGIC;
    outPacket[1] = PACKET_VERSION;
    outPacket[2] = TYPE_METEO;
    outPacket[3] = sizeof(m) & 0xFF;
    outPacket[4] = (sizeof(m) >> 8) & 0xFF;
    memcpy(outPacket + 5, &m, sizeof(m));
    outPacket[5 + sizeof(m)] = pp_calculateXOR(outPacket, 5 + sizeof(m));

    size_t packetSize = 5 + sizeof(m) + 1;

    // Broadcast: fire & forget, nessuna attesa ACK
        bool result = transport->sendBroadcast(outPacket, packetSize);
    
    // DEBUG: verifica se l'invio è riuscito
    LOG_VERBOSE("[SEND] sendBroadcast result: %s, packetSize: %d, TYPE: %d\n", 
             result ? "OK" : "FAIL", packetSize, TYPE_METEO);
    
    return result;
}

/**
 * @brief Invia il pacchetto di salute sensori (Tier 4 ONE-SHOT).
 *        Simula la risposta a un heartbeat TYPE_TIME (cmdEcho = 0x08)
 *        per aggiornare la nodeTable del Gateway.
 *
 * @param deviceId    ID del nodo (DEV_* da devices.h)
 * @param sensorAlive Bitmask salute: bit0=soil, bit1=temp (DS18B20)
 */
bool sendHealthAck(uint8_t deviceId, uint8_t healthMask) {
    if (!transport) return false;

    struct ackData ack;
    ack.deviceID = deviceId;
    ack.status   = 0x01;       // AC_OK
    ack.cmdEcho  = TYPE_TIME;  // 0x08 — simula risposta al heartbeat
    ack.valEcho  = healthMask; // bitmask sensori (bit0=soil, bit1=DS18B20)

    uint8_t outPacket[64];
    size_t packetSize = pp_buildPacket(TYPE_ACK, (uint8_t *)&ack,
                                       sizeof(ackData), outPacket);

    // Piccolo delay per non sovrapporre i pacchetti via radio
    delay(20);
    return transport->sendBroadcast(outPacket, packetSize);
}

} // namespace mqttWifiMessages

#include "sensors.h"
#include "impostazioni.h"
#include <Ds18Manager.h>
#include <MoistureManager.h>
#include <PacketProtocol.h>
#include <shared_config.h>
SystemState stato; // Global system state for health monitoring
namespace sensors {

Ds18Manager ds18(PIN_DS18B20, 0);
MoistureManager soil(PIN_SOIL_PWR, PIN_SOIL_ADC, ADC_DRY_VALUE, ADC_WET_VALUE);

void setup() {
  soil.setup();
#ifdef PIN_DS18B20
  ds18.setup();
#endif
  pinMode(PIN_BATT_ADC, INPUT);

#ifdef PIN_LED
  pinMode(PIN_LED, OUTPUT);
  digitalWrite(PIN_LED, LOW);
#endif
}

bool readSoilMoisture(uint16_t &rawValue, uint16_t &percentValue) {
  return soil.update(rawValue, percentValue);
}

bool readBatteryVoltage(uint16_t &millivolts) {
  uint32_t sum = 0;
  for (int i = 0; i < 10; i++) {
    sum += analogRead(PIN_BATT_ADC);
    delay(5);
  }
  uint16_t rawValue = sum / 10;
  millivolts = map(rawValue, 0, 4095, 0, 3300);
  return true;
}

bool readTemperature(int16_t &temperatureBmp) {
#ifdef PIN_DS18B20
  ds18.update();
  if (stato.sensor_alive[0]) {
    temperatureBmp = PP_ENCODE_TEMP(stato.temps[0]);
    return true;
  }
#endif
  temperatureBmp = 0;
  return false;
}

void blinkLED(int times, int durationMs) {
#ifdef PIN_LED
  for (int i = 0; i < times; i++) {
    digitalWrite(PIN_LED, HIGH);
    delay(durationMs);
    digitalWrite(PIN_LED, LOW);
    if (i < times - 1) delay(durationMs);
  }
#endif
}

} // namespace sensors

Per il problema del "consumo continuo" del partitore, c'è un trucco molto semplice: **usare resistenze dal valore molto alto**.

### Planner Response

È un'idea eccellente e tocca esattamente uno dei limiti fisici del protocollo ESP-NOW.

L'ESP-NOW ha un hard-limit di **20 peer registrati** contemporaneamente (che scendono a 6 se usi la crittografia). Se hai un vivaio con 30-40 vasi, il Gateway andrebbe in crisi se dovesse tenerli tutti in memoria.

Ecco la mia opinione tecnica sui pro e i contro della tua idea di passare al **Broadcast (`FF:FF:FF:FF:FF:FF`)**, e come implementarla al meglio nel tuo ecosistema *Resilient Star*:

### I Vantaggi (Perché è un'ottima idea)
1. **Zero Limiti di Nodi**: Trasmettendo in Broadcast, i nodi C3 non devono registrare il MAC del Gateway, e soprattutto **il Gateway non deve registrare i MAC dei nodi**. Puoi avere 10, 50 o 1000 sensori nel vivaio, per il Gateway è totalmente trasparente.
2. **Zero Overhead di Pairing**: Il sensore si sveglia, "urla" nell'etere il suo dato (Broadcast) e torna a dormire. Non c'è alcun ritardo legato alla risoluzione del MAC.
3. **Meno Collisioni**: Poiché trasmettono ogni 20-30 minuti (e l'invio dura 2 millisecondi), la probabilità statistica che due vasi trasmettano nello stesso identico millisecondo è vicina allo zero.

### L'unico "Contro" e come gestirlo
Quando trasmetti in Unicast, il chip WiFi fa un **ACK Hardware** a bassissimo livello (e te lo notifica in `onDataSent`). 
Se trasmetti in Broadcast, l'hardware NON fa nessun ACK (perché non sa chi dovrebbe rispondere). 
Tuttavia, tu usi già uno **Smart Handshake (Tier 1.5)**, ovvero aspetti un pacchetto logico `TYPE_ACK` dal Gateway. 

### Come adattare l'architettura
Per far funzionare perfettamente questa logica a 40+ nodi, hai due strade per il Gateway quando riceve il dato Broadcast e deve inviare il `TYPE_ACK` per mandare a nanna il sensore C3:

* **Strada 1 (Gateway dinamico - Consigliata)**: Il Gateway riceve il pacchetto dal vivaio, prende il MAC address del mittente (che è sempre incluso nell'header di ricezione), fa al volo un `esp_now_add_peer()`, gli risponde in Unicast con il `TYPE_ACK`, e subito dopo fa `esp_now_del_peer()`. In questo modo la lista peer del Gateway è sempre vuota e il limite dei 20 non viene mai raggiunto.
* **Strada 2 (Doppio Broadcast)**: Anche il Gateway risponde con un `TYPE_ACK` in Broadcast (indirizzato a `FF:FF:FF:FF:FF:FF`). Tutti i sensori lo sentiranno, ma grazie alla tua struttura `ackData` che contiene il `deviceID`, solo il sensore che ha generato il dato accetterà l'ACK, gli altri (se per caso sono svegli) lo scarteranno.
