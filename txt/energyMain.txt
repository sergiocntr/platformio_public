/**
 * @file main.cpp
 * @brief energyMain — Misuratore PZEM-004T via ESP-NOW (PacketProtocol v2.1)
 *
 * Transport: ESP-NOW → EspNowGateway → MQTT (via shared_config.h DEFAULT)
 * Device ID: DEV_ENERGYMAIN_PZEM = 0x40
 */
#include "sensors.h"
#include <Arduino.h>
#include <impostazioni.h>
#include <log_lib.h>
#include <mqttWifi.h>
#include <mqttWifiMessages.h>
#include <myIP.h>

MotivoSpegnimento m_wifi_status =
    (MotivoSpegnimento)254; // default: non aggiornato
uint32_t t_sensors = 0;
const uint32_t time_between_sensors_reads =
    2000; // 2 secondi per campionamento PZEM

void setup() {
  pinMode(buzzerPin, OUTPUT);

delay(1000); // Attesa per stabilizzazione (e.g. seriale, alimentazione)
  rtttl::begin(buzzerPin, startup);

  LOG_INFO("[SYSTEM] EnergyMain v%d avviato (ESP-NOW / PacketProtocol v2.1)", versione);
  Serial.swap(); // Swap Serial su D7(TX)/D8(RX) per liberare D3/D4 al PZEM

  // Stato iniziale: Relay attivi
  daiCorrente.relay('1');
  luceSpia.relay('1');
  
  //Ordine di Inizializzazione secondo specifiche resilient star v3
  mqttWifi::setCallback();

  // Setup trasporto ESP-NOW + handshake ANNOUNCE con gateway
  // DEV_ENERGYMAIN_PZEM (0x40) è passato per il matching ACK e l'Announce
  m_wifi_status = mqttWifi::setupCompleto(
      ipEneMain, EneMainId, PROGETTO_TOPICS, DEV_ENERGYMAIN_PZEM);

  if (m_wifi_status != CONN_OK) {
    LOG_ERROR("[SETUP] Connessione ESP-NOW fallita: %d — riavvio", m_wifi_status);
    delay(5000);
    ESP.restart();
  }


  LOG_INFO("[SYSTEM] Setup completato. Trasporto attivo: ESP-NOW");
}

void loop() {
  // Gestione connessione (ESP-NOW: keepAlive + eventuale fallback WiFi)
  // Include ora il polling automatico dei pacchetti ESP-NOW via mqttWifi::loop()
  mqttWifi::gestisciConnessione();

  // Ciclo lettura sensori ogni 10s
  if (millis() - t_sensors > time_between_sensors_reads) {
    t_sensors = millis();
    sensors::getLocalData();
  }

  delay(10);
}

#include "mqttWifiMessages.h"
#include <PacketProtocol.h>
#include <impostazioni.h>
#include <log_lib.h>
#include <mqttWifi.h>

namespace mqttWifi {

static bool onPacket(const ParsedPacket &pkt);

void callback(char *topic, byte *payload, unsigned int length) {
  LOG_VERBOSE("[CALLBACK] Ricevuto da topic: %s\n", topic);

  // UNIFIED BINARY BUFFER (PacketProtocol)
  if (strcmp(topic, espNowBridgeBuffer) == 0) {
    mqttWifi::pp_dispatchPacket(payload, length);
  }
}

void setCallback() {
  mqttWifi::client.setCallback(callback);
  registerPacketHandler(onPacket);
}

void sendEnePacket(EneMainData dati) {
  dati.deviceID = DEV_ENERGYMAIN_PZEM;
  uint8_t outBuffer[HEADER_SIZE + sizeof(EneMainData) + 1];
  size_t packetSize = pp_buildPacket(TYPE_PZEM, (uint8_t *)&dati,
                                     sizeof(EneMainData), outBuffer);

  if (mqttWifi::publish(espNowBridgeBuffer, outBuffer, (uint16_t)packetSize,
                        false)) {
    LOG_INFO("[SENDPACKET] ✓ Pacchetto PZEM inviato con successo (%d bytes)",
             packetSize);
  } else {
    LOG_ERROR("[SENDPACKET] ✗ Invio pacchetto fallito");
  }
}

static bool onPacket(const ParsedPacket &pkt) {
  switch (pkt.header.type) {
  case TYPE_COMMAND: {
    const cmdData *d = (const cmdData *)pkt.payload;
    if (d->deviceID == CTRL_MAIN_CONTROL) {
      bool on = (d->value == 1);
      daiCorrente.relay(on ? '1' : '0');
      luceSpia.relay(on ? '1' : '0'); // Sincronizza la spia con il carico
      LOG_INFO("[CMD] MAIN_CONTROL set to %s", on ? "ON" : "OFF");
      
      // Invia un ACK binario per confermare il comando
      mqttWifi::sendBinaryAck(CTRL_MAIN_CONTROL, d->command, on);
      
      // Forza l'invio immediato dei dati aggiornati (per vedere subito il cambio di potenza)
      sensors::getLocalData(); 
      return true;
    }
    LOG_WARN("[CMD] Comando ricevuto per deviceID 0x%02X sconosciuto", d->deviceID);
    break;
  }

  case TYPE_TIME: {
    const timeData *d = (const timeData *)pkt.payload;
    LOG_VERBOSE("[TIME] Sync: %02d:%02d", d->hour, d->minute);
    return true; // Sync ricevuto
  }

  default:
    break;
  }
  return false; // Altri tipi non gestiti localmente
}

} // namespace mqttWifi

#include "mqttWifiMessages.h"
#include <cmath>
#include <impostazioni.h>
#include <log_lib.h>
#include <sensors.h>

namespace sensors {
MedianFilter<uint16_t> filter(WINDOW_SIZE);

void getLocalData() {
  // Read the data from the sensor
  float voltage = pzem.voltage();
  float current = pzem.current();
  float power = pzem.power();
  float energy = pzem.energy();
  float frequency = pzem.frequency();
  float pf = pzem.pf();

  // Check if the data is valid
  if (isnan(voltage) || isnan(current) || isnan(power) || isnan(energy) ||
      isnan(frequency) || isnan(pf)) {
    LOG_ERROR("[PZEM] Errore nella lettura dei sensori");
    return;
  }
  
  uint16_t raw_power = uint16_t(power);
  uint16_t filtered_power = filter.AddValue(raw_power);  // Versione filtrata
  
  // Calcola la variazione rispetto all'ultimo valore inviato
  static uint16_t last_sent_power = 0;
  
  // Se è il primo ciclo, inizializza e invia subito
  if (last_sent_power == 0) {
    last_sent_power = filtered_power;
    eneData.e = filtered_power;
    // Invia il primo valore
    eneData.v = uint16_t(voltage * 16);
    eneData.i = uint16_t(current * 128);
    eneData.c = uint16_t(pf * 128);
    LOG_INFO("[PZEM] Primo invio - V:%.1fV I:%.2fA P:%.0fW PF:%.2f", voltage, current, power, pf);
    mqttWifi::sendEnePacket(eneData);
    return;
  }
  
  // Calcola variazione percentuale rispetto all'ultimo valore inviato
  double variazione_percentuale = (filtered_power - last_sent_power) * 100.0 / last_sent_power;

  // Heartbeat: forza invio ogni 5 minuti anche senza variazione sufficiente.
  // Necessario perché con carico stabile il filtro 10% può silenziare il nodo
  // troppo a lungo — Node-RED usa l'ultimo pacchetto come "last seen".
  static uint32_t t_heartbeat = 0;
  const bool heartbeat = (millis() - t_heartbeat >= 300000UL); // 5 min

  // Invia se variazione >= 10% OPPURE heartbeat scaduto
  if (std::abs(variazione_percentuale) >= 10.0 || heartbeat) {
    last_sent_power = filtered_power;
    eneData.e = filtered_power;
    eneData.v = uint16_t(voltage * 16);
    eneData.i = uint16_t(current * 128);
    eneData.c = uint16_t(pf * 128);

    if (heartbeat) {
      t_heartbeat = millis();
      LOG_INFO("[PZEM] Heartbeat 5min - V:%.1fV I:%.2fA P:%.0fW PF:%.2f",
               voltage, current, filtered_power, pf);
    } else {
      LOG_INFO("[PZEM] Variazione %.1f%% - V:%.1fV I:%.2fA P:%.0fW PF:%.2f",
               variazione_percentuale, voltage, current, filtered_power, pf);
    }
    mqttWifi::sendEnePacket(eneData);
  }
}

} // namespace sensors
