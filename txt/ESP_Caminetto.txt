#include "pid_control.h"
#include "impostazioni.h"
#include <PID_v1.h>
#include <EEPROM.h>
#include <log_lib.h>

namespace fanControl {

static double Setpoint, Input, Output;
static PID myPID(&Input, &Output, &Setpoint, 5, 4, 1, REVERSE);

void loadConfig() {
  EEPROM.begin(256);
  EEPROM.get(0, settaggi);
  EEPROM.end();
  LOG_INFO("[PID] Config caricata: T:%u P:%.1f I:%.1f D:%.1f", settaggi.sTemp, settaggi.P, settaggi.I, settaggi.D);
}

void saveConfig() {
  EEPROM.begin(256);
  EEPROM.put(0, settaggi);
  if (EEPROM.commit()) {
    LOG_INFO("[PID] Config salvata");
  } else {
    LOG_ERROR("[PID] Errore salvataggio EEPROM");
  }
  EEPROM.end();
}

void setupPID() {
  // Configurazione PWM ad alta frequenza per evitare ronzii
  analogWriteFreq(30000); 
  pinMode(fanPin, OUTPUT);
  
  // Applica le impostazioni caricate
  myPID.SetOutputLimits(settaggi.sMinFan, settaggi.sMaxFan);
  myPID.SetTunings(settaggi.P, settaggi.I, settaggi.D);
  Setpoint = settaggi.sTemp;
  
  myPID.SetMode(AUTOMATIC);
  LOG_INFO("[PID] Sistema avviato in modalità AUTOMATIC");
}

void updatePID() {
  Input = fireplaceState.tempAmb; // Usiamo la temperatura ambiente (DS18B20)
  myPID.Compute();
  
  // Applica output alla ventola
  analogWrite(fanPin, (int)Output);
  
  // Feedback visivo/MQTT
  fireplaceState.fanPid = (uint8_t)Output; 
}

} // namespace fanControl

#include "temp.h"
#include "impostazioni.h"
#include <DallasTemperature.h>
#include <OneWire.h>
#include <MAX31855.h>
#include <log_lib.h>

OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensorsDS(&oneWire);
MAX31855 sensorsK;

namespace tempSensors {

bool setupSensors() {
  sensorsDS.begin();
  sensorsK.begin(clPin, csPin, doPin);
  LOG_INFO("[TEMP] Sensori Caminetto inizializzati");
  return true;
}

void getLocalData() {
  // 1. Lettura DS18B20 (Ambiente/Circuito)
  sensorsDS.requestTemperatures();
  float t_ds = sensorsDS.getTempCByIndex(0);
  if (t_ds != 85.0 && t_ds != -127.0) {
    fireplaceState.tempAmb = t_ds;
  } else {
    LOG_WARN("[TEMP] Errore lettura DS18B20");
  }

  // 2. Lettura Sonda K (Camino)
  uint8_t status = sensorsK.read();
  if (status == 0) {
    fireplaceState.tempK = sensorsK.getTemperature();
  } else {
    LOG_WARN("[TEMP] Errore lettura Sonda K (Status: %d)", status);
  }

  LOG_VERBOSE("[TEMP] Aria: %.1f°C, Camino: %.1f°C", fireplaceState.tempAmb, fireplaceState.tempK);
}

} // namespace tempSensors

#include <Arduino.h>
#include "impostazioni.h"
#include "mqttWifiMessages.h"
#include "temp.h"

#include <mqttWifi.h>

#include <myIP.h>

static uint32_t t_sens = 0;
static const uint32_t time_between_sensors = 20000; // 20 secondi per campionamento temperature
MotivoSpegnimento m_wifi_status;
void setup() {
  Serial.begin(38400);
  LOG_INFO("[SYSTEM] ESP_Caminetto online (Unified Arch)");

  // Caricamento configurazione PID da EEPROM
  fanControl::loadConfig();

  // Setup WiFi e MQTT (Passiamo DEV_CAMINETTO per handshake binario)
  mqttWifi::setupCompleto(ipCaminetto, CamiId, PROGETTO_TOPICS, DEV_CAMINETTO);
  mqttWifi::setCallback();

  // Inizializzazione sensori (DS18B20 + Sonda K)
  tempSensors::setupSensors();

  // Configurazione PID e Ventola
  fanControl::setupPID();

  // Manda la configurazione attuale via packet all'avvio
  mqttWifi::sendConfig();

  LOG_INFO("[SYSTEM] Setup terminato. Pronto.");
}

void loop() {
  // Gestione automatica della connessione MQTT
  mqttWifi::gestisciConnessione();

  // Calcolo PID continuo per ventilazione
  fanControl::updatePID();

  // Ciclo di campionamento e invio binario MQTT
  if (millis() - t_sens > time_between_sensors) {
    t_sens = millis();
    tempSensors::getLocalData();
    mqttWifi::sendData();
  }

  // Loop MQTT
  mqttWifi::client.loop();
  delay(50);
}

#include "pid_control.h"
#include <mqttWifiMessages.h>

namespace mqttWifi {

// ========== HOOK PACCHETTI PROGETTO ==========
// Gestisce i pacchetti specifici del nodo Caminetto.
// I comandi broadcast (SLEEP, UPDATE, RESET) e TYPE_ACK
// sono già intercettati da mqttWifi::pp_dispatchPacket().
static bool onPacket(const ParsedPacket &pkt); // forward declaration

void setCallback() {
  client.setCallback(callback);
  registerPacketHandler(
      onPacket); // registra l'hook project-specific nel dispatcher
}

static bool onPacket(const ParsedPacket &pkt) {

  switch (pkt.header.type) {

  case TYPE_COMMAND: {
    // CMD_SYS_UPDATE già gestito dalla libreria; qui gestiamo solo comandi
    // rivolti esplicitamente a DEV_CAMINETTO che non sono broadcast.
    // Attualmente nessun comando locale specifico — tenuto per estensibilità.
    const cmdData *d = (const cmdData *)pkt.payload;
    if (d->deviceID == DEV_CAMINETTO) {
      LOG_INFO("[PKT] TYPE_COMMAND per Caminetto ignorato: cmd=0x%02X",
               d->command);
    }
    return true;
  }

  case TYPE_PID_CONFIG: {
    const pidConfigData *d = (const pidConfigData *)pkt.payload;
    if (d->deviceID != DEV_CAMINETTO)
      return false;

    // Anti-eco: scarta pacchetti non originati dal master (Node-RED).
    // Il bit 7 (0x80) dell'updateMask è settato solo da Node-RED.
    if ((d->updateMask & 0x80) == 0) {
      LOG_VERBOSE("[PKT] PID_CONFIG: eco locale scartato (mask=0x%02X)",
                  d->updateMask);
      return true; // consumato, non è un errore
    }

    uint8_t realMask = d->updateMask & 0x7F;
    LOG_INFO("[PKT] PID_CONFIG master (mask=0x%02X)", realMask);

    bool changed = false;
    if ((realMask & 0x01) && settaggi.sTemp != d->sTemp) {
      settaggi.sTemp = d->sTemp;
      changed = true;
    }
    if ((realMask & 0x02) && settaggi.sMinFan != d->sMinFan) {
      settaggi.sMinFan = d->sMinFan;
      changed = true;
    }
    if ((realMask & 0x04) && settaggi.sMaxFan != d->sMaxFan) {
      settaggi.sMaxFan = d->sMaxFan;
      changed = true;
    }
    if ((realMask & 0x08) && settaggi.P != d->P) {
      settaggi.P = d->P;
      changed = true;
    }
    if ((realMask & 0x10) && settaggi.I != d->I) {
      settaggi.I = d->I;
      changed = true;
    }
    if ((realMask & 0x20) && settaggi.D != d->D) {
      settaggi.D = d->D;
      changed = true;
    }

    if (changed) {
      fanControl::saveConfig();
      fanControl::setupPID();
      sendConfig(); // feedback immediato con la nuova configurazione
    } else {
      LOG_VERBOSE("[PKT] PID_CONFIG: parametri invariati, ignoro");
    }
    return true;
  }

  default:
    LOG_WARN("[PKT] PacketType non gestito da Caminetto: 0x%02X\n",
             pkt.header.type);
    return false;
  }
}

// ========== CALLBACK MQTT ==========
// Unica responsabilità: indirizzare i topic al dispatcher.
void callback(char *topic, byte *payload, unsigned int length) {
  LOG_VERBOSE("[CALLBACK] Ricevuto da topic: %s\n", topic);

  if (strcmp(topic, espNowBridgeAck) == 0) {
    // ACK su topic dedicato: gestito direttamente dalla libreria
    handleAckPacket(payload, length);
    return;
  }

  if (strcmp(topic, espNowBridgeBuffer) == 0) {
    mqttWifi::pp_dispatchPacket(payload, length);
  }
}

// ========== INVIO DATI ==========
void sendData() {
  uint8_t buf[HEADER_SIZE + sizeof(caminettoData) + 1];
  caminettoData data;

  data.deviceID = DEV_CAMINETTO;
  data.tempAmb = PP_ENCODE_TEMP(fireplaceState.tempAmb);
  data.tempK = PP_ENCODE_TEMP(fireplaceState.tempK);
  data.fanPid = fireplaceState.fanPid;

  size_t sz = pp_buildPacket(TYPE_CAMINETTO, (uint8_t *)&data,
                             sizeof(caminettoData), buf);
  publish(espNowBridgeBuffer, buf, sz, false);

  LOG_VERBOSE("[SEND] Caminetto: Amb=%.1f K=%.1f Fan=%u",
              fireplaceState.tempAmb, fireplaceState.tempK,
              fireplaceState.fanPid);
}

void sendConfig() {
  uint8_t buf[HEADER_SIZE + sizeof(pidConfigData) + 1];
  pidConfigData data;

  data.deviceID = DEV_CAMINETTO;
  data.updateMask = 0x3F; // tutti i campi validi (senza bit master 0x80)
  data.sTemp = settaggi.sTemp;
  data.sMinFan = settaggi.sMinFan;
  data.sMaxFan = settaggi.sMaxFan;
  data.P = settaggi.P;
  data.I = settaggi.I;
  data.D = settaggi.D;

  size_t sz = pp_buildPacket(TYPE_PID_CONFIG, (uint8_t *)&data,
                             sizeof(pidConfigData), buf);

  AckState res = publishWithAck(espNowBridgeBuffer, buf, sz, 1);
  if (res == END || res == ACK) {
    LOG_INFO("[SEND] Config PID confermata dal server");
  } else {
    LOG_ERROR("[SEND] Config PID non confermata (Timeout)");
  }

  LOG_INFO("[SEND] Config PID: T:%u P:%.1f I:%.1f D:%.1f", settaggi.sTemp,
           settaggi.P, settaggi.I, settaggi.D);
}

} // namespace mqttWifi
